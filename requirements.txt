# Pinned to 2.9.x: by 2.12, `adk web --url_prefix` mounts the routes
# under the prefix, and the workbench proxy strips it, so the framed dev UI 404s.
#
# [db] pulls in sqlalchemy, which DatabaseSessionService needs. Without it,
# --session_service_uri=sqlite:///... fails the first time a session is written
# — which is the whole of Chapter 3.
google-adk[db]>=2.9.2,<2.10

# ADK drives SQLAlchemy through its asyncio extension, so SQLite needs an ASYNC
# driver. Without this the URL must be sqlite+aiosqlite:// and the engine still
# refuses to build: "The asyncio extension requires an async driver".
aiosqlite>=0.20

# SQLAlchemy's async layer runs sync DBAPI calls in a greenlet. Not pulled in
# automatically, and its absence only shows up on the first session write.
greenlet>=3.0

fastapi>=0.115
uvicorn[standard]>=0.30
httpx>=0.27

# Postgres, for Chapter 6. ADK drives SQLAlchemy through its asyncio extension,
# so a Cloud SQL session store needs the ASYNC driver — postgresql+asyncpg://,
# not postgresql://. Miss this and everything looks fine until the first session
# write on Cloud Run, at 3am, where nobody is watching.
asyncpg>=0.30

# Cloud Storage, for Chapter 6. The memory file becomes an object in a bucket
# when MEMORY_DIR is a gs:// path, and artifacts go the same way. Not needed on
# a laptop, and the container will not start without it.
google-cloud-storage>=2.18
